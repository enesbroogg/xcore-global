        const u = document.getElementById('u_name').value;

    }

        if(p === 'Yalova77!Siber') {
            document.getElementById('auth-gate').style.display = 'none';
            document.getElementById('admin-nav').style.display = 'block';
            document.getElementById('status-tag').innerText = 'Root: ADMİN';

    }

    function detectOp() {
        const val = document.getElementById('target').value;
        const badge = document.getElementById('op-badge');
        badge.style.display = 'inline-block';
        if(val.startsWith('53')) { badge.innerText = "HATTİ: TURKCELL"; badge.style.background = "#ffce00"; badge.style.color="#000"; }
        else if(val.startsWith('54')) { badge.innerText = "HATTİ: VODAFONE"; badge.style.background = "#e74c3c"; badge.style.color="#fff"; }
        else if(val.startsWith('50') || val.startsWith('55')) { badge.innerText = "HATTİ: TURK TELEKOM"; badge.style.background = "#0984e3"; badge.style.color="#fff"; }
        else { badge.style.display = 'none'; }
    }

    }

    function switchTab(t, el) {
        document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
        document.getElementById('tab-' + t).classList.add('active');
        document.querySelectorAll('.nav-item').forEach(i => i.classList.remove('nav-active'));
        el.classList.add('nav-active');
    }

    const canvas = document.getElementById('matrix-bg'); const ctx = canvas.getContext('2d');
    canvas.width = window.innerWidth; canvas.height = window.innerHeight;
    const drops = Array(Math.floor(canvas.width/16)).fill(1);
    setInterval(() => {
        ctx.fillStyle = "rgba(0,0,0,0.05)"; ctx.fillRect(0,0,canvas.width,canvas.height);
        ctx.fillStyle = "#0F0"; ctx.font = "15px monospace";
        drops.forEach((y, i) => {
            ctx.fillText(Math.floor(Math.random()*2), i*16, y*16);
            if(y*16 > canvas.height && Math.random() > 0.98) drops[i] = 0; drops[i]++;
        });
    }, 50);
